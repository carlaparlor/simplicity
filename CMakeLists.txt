cmake_minimum_required(VERSION 3.25)
project(simplicity VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(SIMPLICITY_WITH_JUCE "Build the pinned JUCE adapter and headless bootstrap" ON)
option(SIMPLICITY_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(SIMPLICITY_COVERAGE "Instrument first-party code for gcov" OFF)
if(SIMPLICITY_SANITIZERS AND SIMPLICITY_COVERAGE)
  message(FATAL_ERROR "Use separate sanitizer and coverage build trees")
endif()
if(SIMPLICITY_SANITIZERS OR SIMPLICITY_COVERAGE)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Instrumentation requires GCC or Clang")
  endif()
endif()
if(SIMPLICITY_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(SIMPLICITY_COVERAGE)
  add_compile_options(--coverage -O0 -g)
  add_link_options(--coverage)
endif()
function(simplicity_strict target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${PROJECT_SOURCE_DIR}/tools/check_architecture.py
  WORKING_DIRECTORY ${PROJECT_SOURCE_DIR} COMMAND_ERROR_IS_FATAL ANY)
enable_testing()
add_subdirectory(modules/core)
add_subdirectory(modules/sdk)
add_subdirectory(modules/engine)
if(SIMPLICITY_WITH_JUCE)
  include(FetchContent)
  # JUCE 8.0.12, immutable upstream commit. See docs/dependencies.md for licensing.
  FetchContent_Declare(JUCE
    GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
    GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850
    GIT_SHALLOW FALSE
    SOURCE_SUBDIR simplicity-no-upstream-project)
  # Populate without configuring all upstream modules (which probe GUI packages).
  FetchContent_MakeAvailable(JUCE)
  include(${juce_SOURCE_DIR}/extras/Build/CMake/JUCEModuleSupport.cmake)
  juce_add_modules(ALIAS_NAMESPACE juce
    ${juce_SOURCE_DIR}/modules/juce_core
    ${juce_SOURCE_DIR}/modules/juce_audio_basics)
  add_subdirectory(modules/juce_adapter)
  add_subdirectory(app)
endif()
add_subdirectory(tests)
